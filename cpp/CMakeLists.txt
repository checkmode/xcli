cmake_minimum_required(VERSION 3.22)
project(xray_gateway_panel VERSION 1.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(XG_SANITIZE "Build with address and undefined-behavior sanitizers" OFF)
find_package(PkgConfig REQUIRED)
find_package(Protobuf REQUIRED)
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.11 REQUIRED)
pkg_check_modules(HTTP REQUIRED IMPORTED_TARGET cpp-httplib)
pkg_check_modules(SQLITE REQUIRED IMPORTED_TARGET sqlite3)
pkg_check_modules(CRYPTO REQUIRED IMPORTED_TARGET libcrypto)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(QR REQUIRED IMPORTED_TARGET libqrencode)
pkg_check_modules(PNG REQUIRED IMPORTED_TARGET libpng)
pkg_check_modules(GRPC REQUIRED IMPORTED_TARGET grpc++)
find_program(GRPC_PLUGIN grpc_cpp_plugin REQUIRED)

file(GLOB_RECURSE PROTO_FILES CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/proto" "proto/*.proto")
list(FILTER PROTO_FILES EXCLUDE REGEX "(^|/)\\._")
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED}")
set(PROTO_SOURCES)
foreach(PROTO IN LISTS PROTO_FILES)
  string(REGEX REPLACE "\\.proto$" ".pb.cc" SOURCE "${PROTO}")
  string(REGEX REPLACE "\\.proto$" ".pb.h" HEADER "${PROTO}")
  add_custom_command(OUTPUT "${GENERATED}/${SOURCE}" "${GENERATED}/${HEADER}"
    COMMAND protobuf::protoc --cpp_out=${GENERATED} -I ${CMAKE_CURRENT_SOURCE_DIR}/proto ${PROTO}
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/proto/${PROTO}"
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/proto" VERBATIM)
  list(APPEND PROTO_SOURCES "${GENERATED}/${SOURCE}")
endforeach()
foreach(PROTO app/proxyman/command/command.proto app/stats/command/command.proto)
  string(REGEX REPLACE "\\.proto$" ".grpc.pb.cc" SOURCE "${PROTO}")
  string(REGEX REPLACE "\\.proto$" ".grpc.pb.h" HEADER "${PROTO}")
  add_custom_command(OUTPUT "${GENERATED}/${SOURCE}" "${GENERATED}/${HEADER}"
    COMMAND protobuf::protoc --grpc_out=${GENERATED} --plugin=protoc-gen-grpc=${GRPC_PLUGIN}
      -I ${CMAKE_CURRENT_SOURCE_DIR}/proto ${PROTO}
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/proto/${PROTO}"
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/proto" VERBATIM)
  list(APPEND PROTO_SOURCES "${GENERATED}/${SOURCE}")
endforeach()
add_library(xray_proto STATIC ${PROTO_SOURCES})
target_include_directories(xray_proto PUBLIC "${GENERATED}")
target_link_libraries(xray_proto PUBLIC protobuf::libprotobuf PkgConfig::GRPC)

add_library(xg_backend STATIC src/database.cpp src/crypto.cpp src/backend.cpp src/security.cpp src/maintenance.cpp src/operations.cpp src/webhook.cpp src/xray.cpp src/http.cpp src/runtime.cpp)
target_include_directories(xg_backend PUBLIC include)
target_link_libraries(xg_backend PUBLIC xray_proto nlohmann_json::nlohmann_json PkgConfig::HTTP
  PkgConfig::SQLITE PkgConfig::CRYPTO PkgConfig::CURL PkgConfig::QR PkgConfig::PNG Threads::Threads)
target_compile_options(xg_backend PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wformat=2)
target_compile_definitions(xg_backend PUBLIC XG_VERSION="${PROJECT_VERSION}")
execute_process(COMMAND git -C "${CMAKE_CURRENT_SOURCE_DIR}" rev-parse --short=12 HEAD OUTPUT_VARIABLE XG_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT XG_REVISION MATCHES "^[0-9a-f]+$")
  set(XG_REVISION "unknown")
else()
  execute_process(COMMAND git -C "${CMAKE_CURRENT_SOURCE_DIR}" diff --quiet HEAD RESULT_VARIABLE XG_DIRTY ERROR_QUIET)
  if(XG_DIRTY EQUAL 1)
    string(APPEND XG_REVISION "-dirty")
  endif()
endif()
target_compile_definitions(xg_backend PUBLIC XG_REVISION="${XG_REVISION}")
add_executable(xray-gateway-panel src/main.cpp)
target_link_libraries(xray-gateway-panel PRIVATE xg_backend)
add_executable(xg-unit-tests tests/unit.cpp)
target_link_libraries(xg-unit-tests PRIVATE xg_backend)
add_executable(xg-operations-tests tests/operations.cpp)
target_link_libraries(xg-operations-tests PRIVATE xg_backend)
add_executable(xg-http-queue-tests tests/http_queue.cpp)
add_executable(xg-runtime-tests tests/runtime.cpp)
target_link_libraries(xg-runtime-tests PRIVATE xg_backend)
target_include_directories(xg-http-queue-tests PRIVATE include)
target_link_libraries(xg-http-queue-tests PRIVATE PkgConfig::HTTP PkgConfig::CRYPTO Threads::Threads)
if(XG_SANITIZE)
  foreach(TARGET xg_backend xray-gateway-panel xg-unit-tests xg-operations-tests xg-http-queue-tests xg-runtime-tests)
    target_compile_options(${TARGET} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${TARGET} PUBLIC -fsanitize=address,undefined)
  endforeach()
endif()
enable_testing()
add_test(NAME backend-unit COMMAND xg-unit-tests)
add_test(NAME operations-unit COMMAND xg-operations-tests)
add_test(NAME runtime-unit COMMAND xg-runtime-tests)
add_test(NAME http-queue-bound COMMAND xg-http-queue-tests)
set_tests_properties(http-queue-bound PROPERTIES TIMEOUT 15)
install(TARGETS xray-gateway-panel RUNTIME DESTINATION bin)
